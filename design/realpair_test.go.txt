package plan_test

import (
	"os"
	"path/filepath"
	"strings"
	"testing"

	"github.com/dlaszlo/camp/internal/config"
	"github.com/dlaszlo/camp/internal/pathx"
	"github.com/dlaszlo/camp/internal/plan"
)

// The pair camp is actually judged on: ~/dev/diet-coach as the code
// repository and ~/dev/diet-coach-workspace as the workspace.
//
// Before the migration these two share names at their roots, and the gate
// has to stop the composition and say so in a message that names both
// sides. This test is read-only -- it runs the gate against the real
// directories and mounts nothing -- and it skips itself where those
// directories are not present, because it is an acceptance check for one
// machine and not a portable unit test.
func TestRealPairPreMigrationIsBlockedByTheGate(t *testing.T) {
	home, err := os.UserHomeDir()
	if err != nil {
		t.Skip("no home directory")
	}
	env := filepath.Join(home, "dev")
	code := filepath.Join(env, "diet-coach")
	workspace := filepath.Join(env, "diet-coach-workspace")
	for _, path := range []string{code, workspace} {
		if _, err := os.Stat(path); err != nil {
			t.Skipf("%s is not on this machine", path)
		}
	}

	yaml := `env: ` + env + `
merged: diet-coach-live
repositories:
  - { name: workspace, path: diet-coach-workspace }
  - { name: code,      path: diet-coach }
overlayfs: { lower: [workspace], upper: code }
allow_overlap: [.gitignore]
steps:
  - mount_rw:
      - { source: "code/.git", target: ".git" }
  - git_exclude
`
	cfg, err := config.Parse([]byte(yaml), config.Path(env))
	if err != nil {
		t.Fatalf("the target configuration did not parse:\n%v", err)
	}

	lowerRoot, err := pathx.ReadDirBeneath(workspace, nil)
	if err != nil {
		t.Fatalf("listing %s: %v", workspace, err)
	}
	upperRoot, err := pathx.ReadDirBeneath(code, nil)
	if err != nil {
		t.Fatalf("listing %s: %v", code, err)
	}

	refused := plan.Gate(cfg, lowerRoot, upperRoot)
	if refused.Empty() {
		t.Skip("the two repositories no longer overlap: the migration has " +
			"happened, and there is nothing left for this check to catch")
	}

	t.Logf("the gate refuses the pre-migration pair, %d overlap(s):\n\n%s",
		len(refused), refused.Error())

	message := refused.Error()
	for _, want := range []string{workspace, code, "allow_overlap"} {
		if !strings.Contains(message, want) {
			t.Errorf("the refusal does not mention %q; it has to name both sides "+
				"and both ways out", want)
		}
	}
	if strings.Contains(message, "Overlap detected") {
		t.Error("the message has to say more than that an overlap exists")
	}
}
